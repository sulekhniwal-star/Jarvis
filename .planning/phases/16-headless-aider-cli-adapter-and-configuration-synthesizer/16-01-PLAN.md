  <name>Implement Aider Configuration Synthesizer</name>
  <files>
    src/lib/agents/engine/aider/config-synthesizer.ts
  </files>
  <action>
    Create `src/lib/agents/engine/aider/config-synthesizer.ts`:
    1. Implement `synthesizeAiderArgs(context: CodingTaskContext, options?: Partial<AiderCliOptions>): string[]`:
       - Base flags: `["--yes-always", "--no-show-model-warnings", "--no-analytics", "--no-check-update"]`
       - Model mapping: `--model <model>`, `--openai-api-base <apiBase>` if OpenRouter
       - Pass read-only context files via `--read <file>`
       - Pass instructions via `--message <instructions>` or `--message-file <path>`
       - Pass strictly allowed target files as positional arguments (from `context.allowedFilePatterns`)
    2. Implement `generateAiderYamlConfig(context: CodingTaskContext, options?: Partial<AiderCliOptions>): string`:
       - Generates `.aider.conf.yml` content formatted with model, auto-commits policy, and file read constraints.
    3. Export `AiderConfigSynthesizer` class and helper functions.
  </action>
  <verify>
    Run `pnpm.cmd exec tsc --noEmit` to verify synthesizer logic.
  </verify>
  <done>
    Aider command line arguments and YAML configuration are dynamically synthesized from CodingTaskContext.
  </done>
</task>

<task type="auto">
  <name>Implement Aider Output Parser</name>
  <files>
    src/lib/agents/engine/aider/output-parser.ts
  </files>
  <action>
    Create `src/lib/agents/engine/aider/output-parser.ts`:
    1. Implement `parseAiderOutput(result: AiderProcessResult, targetFiles?: string[]): ParsedAiderOutput`:
       - Extracts commit SHA: searches stdout for `Commit ([0-9a-f]{7,40})` or `git commit ([0-9a-f]{7,40})`.
       - Extracts unified diff if present in stdout.
       - Identifies modified files mentioned in output or matching target files.
       - Detects failure patterns (syntax errors, provider rate limits, rejection messages).
    2. Implement `extractExecutionArtifacts(result: AiderProcessResult, parsed: ParsedAiderOutput): ExecutionArtifact[]`:
       - Generates `build_log` artifact from stdout.
       - Generates `test_log` artifact if test commands were executed.
       - Generates `diff` artifact if diff patch was captured.
    3. Export `AiderOutputParser` class and helper functions.
  </action>
  <verify>
    Run `pnpm.cmd exec tsc --noEmit` to verify output parsing logic.
  </verify>
  <done>
    Aider process output parser extracts commit SHAs, diffs, modified files, and formats execution artifacts.
  </done>
</task>

<task type="auto">
  <name>Implement Plan Decision API Route Handler</name>
  <files>
    src/app/api/tasks/[id]/plan/decision/route.ts
  </files>
  <action>
    Author `src/app/api/tasks/[id]/plan/decision/route.ts`:
    - `POST` route handler accepting body `{ decision: 'approve' | 'request_revision', feedback?: string, userId?: string }`.
    - Validates task exists and status is `planned` or `awaiting_approval`.
    - Fetches latest plan from `PlanRepository`.
    - If `decision === 'approve'`:
      - Updates plan status to `approved`.
      - If plan required approvals, records approval via `ApprovalRepository.create(...)`.
      - Transitions task status to `running` via `TaskRepository.updateStatus(taskId, 'running')`.
      - Returns 200 with `{ taskId, planId, status: 'approved', taskStatus: 'running' }`.
    - If `decision === 'request_revision'`:
      - Updates plan status to `superseded`.
      - Transitions task status to `specifying` via `TaskRepository.updateStatus(taskId, 'specifying')`.
      - Returns 200 with `{ taskId, planId, status: 'superseded', taskStatus: 'specifying', feedback }`.
  </action>
</task>
